int isEven(int n)
{
    if (n = 0)
    {
        return 1;
    }
    return isOdd(n - 1);
}

int isOdd(int n)
{
    if (n = 0)
    {
        return 0;
    }
    return isEven(n - 1);
}

void main()
{
    PrintInt(isEven(4));
    PrintInt(isOdd(4));
    PrintInt(isEven(7));
    PrintInt(isOdd(7));
}

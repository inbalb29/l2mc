class Node
{
    int val := 0;
    Node next := nil;
}
void main()
{
    Node n3 := new Node;
    n3.val := 3;
    Node n2 := new Node;
    n2.val := 2;
    n2.next := n3;
    Node n1 := new Node;
    n1.val := 1;
    n1.next := n2;
    int count := 0;
    Node curr := n1;
    while ((curr = nil) = 0)
    {
        count := count + 1;
        curr := curr.next;
    }
    PrintInt(count);
}

class Point
{
    int x := 0;
    int y := 0;
}
class Geom
{
    int distSq(Point a, Point b)
    {
        int dx := a.x - b.x;
        int dy := a.y - b.y;
        return dx * dx + dy * dy;
    }
}

void main()
{
    Point p := new Point;
    Point q := new Point;
    p.x := 3;
    p.y := 0;
    q.x := 0;
    q.y := 4;
    Geom g := new Geom;
    PrintInt(g.distSq(p, q));
}

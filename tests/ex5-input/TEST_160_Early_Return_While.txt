array IA = int[];
int findGtTwo(IA arr, int len)
{
    int i := 0;
    while (i < len)
    {
        if (arr[i] > 2)
        {
            return arr[i];
        }
        i := i + 1;
    }
    return 0;
}
void main()
{
    IA a := new int[5];
    a[0] := 1;
    a[1] := 2;
    a[2] := 3;
    a[3] := 4;
    a[4] := 5;
    PrintInt(findGtTwo(a, 5));
}

class Counter
{
    int count := 0;
    void inc()
    {
        count := count + 1;
    }
    void incTwice()
    {
        inc();
        inc();
    }
    int getCount()
    {
        return count;
    }
}

void main()
{
    Counter c := new Counter;
    c.incTwice();
    PrintInt(c.getCount());
    c.inc();
    PrintInt(c.getCount());
}

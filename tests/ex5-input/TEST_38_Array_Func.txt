array IntArray = int[];

void fill(IntArray arr, int size)
{
    int i := 0;
    while (i < size)
    {
        arr[i] := i * i;
        i := i + 1;
    }
}

int sum(IntArray arr, int size)
{
    int i := 0;
    int s := 0;
    while (i < size)
    {
        s := s + arr[i];
        i := i + 1;
    }
    return s;
}

void main()
{
    IntArray a := new int[5];
    fill(a, 5);
    PrintInt(sum(a, 5));
    PrintInt(a[3]);
}
